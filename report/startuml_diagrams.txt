DIAGRAM 7: Class Diagram
============================================================

@startuml
skinparam classAttributeIconSize 0

enum Platform {
  facebook
  instagram
  tiktok
  youtube
  snapchat
}

enum VideoStatus {
  uploaded
  processing
  ready
  failed
}

enum PostStatus {
  scheduled
  posted
  failed
}

enum AlertType {
  error
  copyright
  success
}

enum AlertStatus {
  unresolved
  resolved
}

enum UserRole {
  admin
  member
}

class User {
  +id: UUID <<PK>>
  +email: VARCHAR(255) <<UNIQUE>>
  +fullName: VARCHAR(120)
  +role: UserRole
  +createdAt: TIMESTAMP
  +updatedAt: TIMESTAMP
}

class Account {
  +id: UUID <<PK>>
  +platform: Platform
  +accountName: VARCHAR(255)
  +accessToken: TEXT <<AES-256>>
  +refreshToken: TEXT
  +tokenExpiry: TIMESTAMP
  +userId: UUID <<FK>>
  +metadata: JSONB
  +createdAt: TIMESTAMP
}

class Video {
  +id: UUID <<PK>>
  +userId: UUID <<FK>>
  +originalUrl: TEXT
  +processedUrl: TEXT
  +duration: INTEGER
  +status: VideoStatus
  +createdAt: TIMESTAMP
}

class Post {
  +id: UUID <<PK>>
  +videoId: UUID <<FK>>
  +accountId: UUID <<FK>>
  +platform: Platform
  +scheduledAt: TIMESTAMP
  +postedAt: TIMESTAMP
  +status: PostStatus
  +externalPostId: VARCHAR(255)
  +errorMessage: TEXT
  +metadata: JSONB
  +createdAt: TIMESTAMP
}

class Analytics {
  +id: UUID <<PK>>
  +postId: UUID <<FK>>
  +views: INTEGER
  +likes: INTEGER
  +comments: INTEGER
  +shares: INTEGER
  +revenue: NUMERIC(12,2)
  +fetchedAt: TIMESTAMP
}

class Alert {
  +id: UUID <<PK>>
  +type: AlertType
  +status: AlertStatus
  +message: TEXT
  +platform: Platform
  +accountId: UUID <<FK>>
  +postId: UUID <<FK>>
  +createdAt: TIMESTAMP
}

User "1" --> "0..*" Account : owns
User "1" --> "0..*" Video : uploads
Account "1" --> "0..*" Post : publishes through
Video "1" --> "0..*" Post : is posted as
Post "1" --> "0..*" Analytics : has metrics
Account "0..1" <-- "0..*" Alert : references
Post "0..1" <-- "0..*" Alert : references
@enduml

DIAGRAM 8: Sequence Diagram — OAuth Account Connection
============================================================

@startuml
actor "Content Creator" as U
participant "Dashboard\n(React)" as D
participant "API Server\n(Fastify)" as API
participant "Platform\nOAuth Server" as P
database "PostgreSQL\n(Supabase)" as DB

U -> D : Click "Connect YouTube"
D -> API : GET /api/v1/accounts/youtube/connect
API -> API : Generate OAuth URL\n+ state + scopes
API --> D : 302 Redirect to OAuth URL
D -> P : Redirect to consent screen
U -> P : Grant permissions
P --> D : Redirect to callback\nwith authorization code
D -> API : GET /api/v1/accounts/youtube/callback?code=xxx
API -> P : POST token exchange\n(code → access_token + refresh_token)
P --> API : Return tokens
API -> API : AES-256 encrypt tokens
API -> DB : INSERT INTO accounts\n(platform, tokens, userId)
DB --> API : Account created
API --> D : 302 Redirect to dashboard\n+ success status
D --> U : Show "YouTube Connected"
@enduml

DIAGRAM 9: Sequence Diagram — Schedule & Publish Post
============================================================

@startuml
actor "Content Creator" as U
participant "Dashboard\n(React)" as D
participant "API Server\n(Fastify)" as API
database "PostgreSQL" as DB
participant "Redis\n(BullMQ)" as Q
participant "Worker\nProcess" as W
participant "Platform API\n(YouTube/TikTok/Meta)" as P

U -> D : Create new post\n(video, platform, time)
D -> API : POST /api/v1/posts
API -> DB : INSERT post\n(status=scheduled)
DB --> API : Post created
API -> Q : Enqueue delayed job\n(postId, scheduledAt)
Q --> API : Job enqueued
API --> D : 201 Created
D --> U : Show post in calendar

... time passes until scheduledAt ...

Q -> W : Dequeue job at scheduled time
W -> DB : SELECT post + account details
DB --> W : Post & account data
W -> W : Decrypt access token
W -> P : Upload video + metadata\nusing OAuth token
alt Success
  P --> W : Return externalPostId
  W -> DB : UPDATE post\n(status=posted, externalPostId)
  W --> Q : Job complete
else Failure
  P --> W : Error response
  W -> DB : UPDATE post\n(status=failed, errorMessage)
  W -> DB : INSERT alert\n(type=error)
  W --> Q : Job failed
end
@enduml

DIAGRAM 10: Sequence Diagram — AI Video Enhancement
============================================================

@startuml
actor "Content Creator" as U
participant "AI Studio\n(React)" as D
participant "Enhancement\nPipeline" as P
participant "FFmpeg" as FF
participant "Quality\nAnalyzer" as QA
participant "Real-ESRGAN" as SR
participant "GFPGAN" as GF

U -> D : Upload video file
D -> P : upload_video(file)
P -> P : Validate format & size
P -> FF : Extract frames\n(ffmpeg → frame_%06d.png)
FF --> P : Frames extracted
P -> FF : Extract audio track
FF --> P : audio.aac
P -> QA : analyze_quality(frames)
QA -> QA : Detect blur, noise,\nlow-light, faces
QA --> P : Quality Report\n(needs_sr, needs_denoise,\nfaces_detected)
P -> P : Apply classical filters\n(bilateral, Gaussian, unsharp)
alt needs_sr = true
  P -> SR : Super-resolve frames\n(2x or 4x upscale)
  SR --> P : Enhanced frames
end
alt faces_detected = true
  P -> GF : Face restoration\n(GFPGAN v1.4)
  GF --> P : Restored frames
end
P -> FF : Reassemble video\n(H.264, CRF 18, faststart)
FF --> P : Output video
P -> P : Compute metrics\n(PSNR, SSIM, LPIPS)
P -> P : Generate publish manifest
P --> D : Enhancement complete\n+ metrics + manifest
D --> U : Show before/after preview\n+ quality metrics
@enduml

DIAGRAM 13: AI Pipeline Activity Diagram
============================================================

@startuml
start
:Upload Video;
:Validate format & size;
if (Valid?) then (yes)
  :Extract frames (FFmpeg);
  :Extract audio track;
  :Analyze quality\n(blur, noise, low-light, faces);

  if (Needs denoise?) then (yes)
    :Apply bilateral filter;
  else (no)
    :Apply balanced filter;
  endif

  if (Resolution < 1080p?) then (yes)
    :Run Real-ESRGAN\nsuper-resolution (2x/4x);
  else (no)
  endif

  if (Faces detected?) then (yes)
    :Run GFPGAN\nface restoration;
  else (no)
  endif

  :Reassemble video\n(H.264, CRF 18, faststart);
  :Compute quality metrics\n(PSNR, SSIM, LPIPS);
  :Generate publish manifest;
  :Show before/after preview;
  stop
else (no)
  :Show error message;
  stop
endif
@enduml
